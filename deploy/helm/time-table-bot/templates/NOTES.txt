Create secrets before install:

  kubectl create secret generic {{ .Values.env.telegramBotToken.secretName }} \
    --from-literal={{ .Values.env.telegramBotToken.secretKey }}='<telegram-bot-token>' \
    --from-literal={{ .Values.postgres.passwordSecret.key }}='<postgres-password>'

Optional: add Qwen Cloud API key to enable natural-language booking:

  kubectl create secret generic qwen-model-studio \
    --from-literal=QWEN_API_KEY='<qwen-api-key>' \
    --from-literal=QWEN_BASE_URL='https://dashscope-intl.aliyuncs.com/compatible-mode/v1'

  helm upgrade --install {{ .Release.Name }} ./deploy/helm/time-table-bot \
    --namespace {{ .Release.Namespace }} \
    --set env.qwen.apiKey.secretName=qwen-model-studio \
    --set env.qwen.apiKey.secretKey=QWEN_API_KEY \
    --set env.qwen.baseUrlSecret.secretName=qwen-model-studio \
    --set env.qwen.baseUrlSecret.secretKey=QWEN_BASE_URL

Optional: if you want to pass DATABASE_URL as a full secret value:

  kubectl create secret generic my-db-url-secret \
    --from-literal=database-url='postgres://user:pass@postgres:5432/db?sslmode=disable'

  helm install {{ .Release.Name }} ./deploy/helm/time-table-bot \
    --namespace {{ .Release.Namespace }} \
    --set env.databaseUrl.secretName=my-db-url-secret \
    --set env.databaseUrl.secretKey=database-url

Install with generated DATABASE_URL from POSTGRES_*:

  helm install {{ .Release.Name }} ./deploy/helm/time-table-bot --namespace {{ .Release.Namespace }}

Render templates locally (without cluster access):

  helm template {{ .Release.Name }} ./deploy/helm/time-table-bot --namespace {{ .Release.Namespace }}
